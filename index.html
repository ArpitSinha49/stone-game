How To Play

Stone Pickup (Ancient Nim)

1.

Secret Stone Mound: The initial number of stones is kept secret and hidden from all players. The maximum possible stone count increases with the total number of players in the game.

2.

Player Order: The player added first takes the first move. Turns proceed strictly in the order players joined the lobby.

3.

Stone Pick Rule: On your turn, you must take 1, 2, or 3 stones from the pile. If fewer stones remain than your choice, taking them claims all remaining stones.

4.

The Victory Condition: Whosoever takes the very last stone claims victory and triumphs as the champion!

Python Flask Backend & VS Code Guide

Ready-to-run REST API architecture for VS Code

⚡ How to run in VS Code in 3 minutes:

  1. Open VS Code and create a new folder named stone-game.
  2. Open your VS Code Integrated Terminal (Ctrl + ` or Cmd + `).
  3. Create & activate a virtual environment:
    python -m venv venv
    # Windows: venv\Scripts\activate
    # Mac/Linux: source venv/bin/activate
  4. Install Flask and Flask-CORS:
    pip install flask flask-cors
  5. Create a file named app.py, paste the code below, and run:
    python app.py
app.py (Flask REST API Server with Hidden Pile) 
import random
from flask import Flask, request, jsonify
from flask_cors import CORS

app = Flask(__name__)
CORS(app)  # Enables cross-origin requests

# Server-side authoritative state (exact pile count is never leaked to players)
game_state = {
    "status": "lobby",  # lobby | active | finished
    "players": [],      # player names in turn order
    "current_turn": 0,
    "secret_stones": 0, # HIDDEN on server
    "initial_stones": 0,
    "round_number": 1,
    "winner": None,
    "history": []
}

def get_public_state():
    """Returns state with stone counts hidden."""
    rem = game_state["secret_stones"]
    # Qualitative hint based on remaining stones
    if rem > 25:
        hint = "Hefty Mound"
    elif rem > 12:
        hint = "Substantial Heap"
    elif rem > 5:
        hint = "Dwindling Cluster"
    else:
        hint = "A Few Stones Remain"

    return {
        "status": game_state["status"],
        "players": game_state["players"],
        "current_turn": game_state["current_turn"],
        "round_number": game_state["round_number"],
        "pile_vibe": hint if game_state["status"] == "active" else "Concealed",
        "winner": game_state["winner"],
        "history": game_state["history"],
        # Disclose actual count ONLY once the match finishes
        "revealed_initial": game_state["initial_stones"] if game_state["status"] == "finished" else None
    }

@app.route("/api/state", methods=["GET"])
def get_state():
    return jsonify(get_public_state())

@app.route("/api/players", methods=["POST"])
def add_player():
    if game_state["status"] != "lobby":
        return jsonify({"error": "Cannot add players mid-game."}), 400
    
    data = request.get_json() or {}
    name = data.get("name", "").strip()
    if not name:
        return jsonify({"error": "Player name cannot be empty."}), 400
    if name in game_state["players"]:
        return jsonify({"error": "Player already in circle."}), 400

    game_state["players"].append(name)
    return jsonify(get_public_state())

@app.route("/api/start", methods=["POST"])
def start_game():
    n = len(game_state["players"])
    if n < 2:
        return jsonify({"error": "Need at least 2 players to start."}), 400

    # Max stone count scales with total players; exact initial amount is kept secret!
    stones = random.randint(15, 15 * n)
    game_state["status"] = "active"
    game_state["current_turn"] = 0
    game_state["secret_stones"] = stones
    game_state["initial_stones"] = stones
    game_state["round_number"] = 1
    game_state["winner"] = None
    game_state["history"] = [{
        "action": "start",
        "message": f"Match started across {n} players. The sacred mound is sealed."
    }]

    return jsonify(get_public_state())

@app.route("/api/move", methods=["POST"])
def make_move():
    if game_state["status"] != "active":
        return jsonify({"error": "No active match."}), 400

    data = request.get_json() or {}
    pick = data.get("count")
    if pick not in [1, 2, 3]:
        return jsonify({"error": "Choice must be 1, 2, or 3 stones."}), 400

    current_player = game_state["players"][game_state["current_turn"]]
    actual_removed = min(pick, game_state["secret_stones"])
    game_state["secret_stones"] -= actual_removed

    if game_state["secret_stones"] <= 0:
        game_state["status"] = "finished"
        game_state["winner"] = current_player
        game_state["history"].append({
            "player": current_player,
            "removed": actual_removed,
            "message": f"🏆 {current_player} took the final stone(s) and WON!"
        })
    else:
        game_state["history"].append({
            "player": current_player,
            "removed": actual_removed,
            "message": f"{current_player} drew from the secret mound."
        })
        game_state["current_turn"] = (game_state["current_turn"] + 1) % len(game_state["players"])
        if game_state["current_turn"] == 0:
            game_state["round_number"] += 1

    return jsonify(get_public_state())

if __name__ == "__main__":
    print("✨ Stone Game Server running at http://localhost:5000 (Secret Pile Enabled)")
    app.run(host="0.0.0.0", port=5000, debug=True)
